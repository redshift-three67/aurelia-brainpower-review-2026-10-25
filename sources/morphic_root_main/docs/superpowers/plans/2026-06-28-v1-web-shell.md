# Morphic V1 Web Shell Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a small Radi144-inspired FastAPI Web Shell for Morphic Engine V1 that exposes system health, source registry, DSGVO-safe client views and a Top-N scan preview without changing the core engines.

**Architecture:** The web layer is a thin adapter under `apps/web`. Static HTML/CSS/JS renders the first operator screen, while JSON endpoints reuse existing services from `core.db.health`, `core.source_registry` and `core.compliance.runtime_manifest`. The scan endpoint only previews the canonical CLI command in this slice; it does not execute long-running scans.

**Tech Stack:** FastAPI, Starlette static files, Pydantic, existing Morphic core services, pytest with FastAPI `TestClient`.

**Execution note:** The local Starlette `TestClient` requires an unavailable `httpx2` package in this environment. The implemented test contract therefore uses a tiny inline ASGI request helper in `tests/test_web_shell.py` and keeps the same behavioral assertions without adding a new dependency.

```python
from pathlib import Path

from fastapi import FastAPI
from fastapi.responses import FileResponse
from fastapi.staticfiles import StaticFiles
from pydantic import BaseModel, Field

from core.compliance.runtime_manifest import RuntimeManifestService
from core.db.health import collect_runtime_health
from core.source_registry import SourceRegistry


STATIC_DIR = Path(__file__).with_name("static")


class ScanPreviewRequest(BaseModel):
    last_name: str = Field(min_length=1)
    first_name: str = Field(min_length=1)
    birth_date: str = Field(min_length=1)
    birth_place: str = Field(min_length=1)
    focus: str = Field(min_length=1)
    top_n: int = Field(default=5, ge=3, le=9)
    export: str = "Benutzer_Archive"
    hardware_profile: str = "windows"
    k_total: int = 4096
    top_k: int = 384
    synergy: bool = True
    synergy_combos: int = 4
    synergy_threshold: float = 6.0


def create_app(runtime_dir: str | Path | None = None) -> FastAPI:
    app = FastAPI(title="Morphic Engine V1 Web Shell")
    app.state.runtime_dir = Path(runtime_dir) if runtime_dir is not None else None
    app.mount("/static", StaticFiles(directory=STATIC_DIR), name="static")

    @app.get("/")
    def index() -> FileResponse:
        return FileResponse(STATIC_DIR / "index.html")

    @app.get("/api/health")
    def health() -> dict:
        return collect_runtime_health()

    @app.get("/api/sources")
    def sources() -> dict:
        registry = SourceRegistry.load_default()
        items = [source.to_mapping() for source in registry.list_sources()]
        status_counts = {"active": 0, "disabled": 0, "quarantined": 0, "deprecated": 0}
        for source in registry.list_sources():
            status_counts[source.status] = status_counts.get(source.status, 0) + 1
        return {"count": len(items), "status_counts": status_counts, "sources": items}

    @app.get("/api/privacy/clients")
    def privacy_clients() -> dict:
        service = RuntimeManifestService(app.state.runtime_dir)
        clients = service.list_clients_safe()
        return {"count": len(clients), "clients": clients}

    @app.post("/api/scan/preview")
    def scan_preview(request: ScanPreviewRequest) -> dict:
        command = [
            "python",
            "-m",
            "apps.cli.main",
            "scan",
            "top3-scan",
            request.last_name,
            request.first_name,
            "--birth-date",
            request.birth_date,
            "--birth-place",
            request.birth_place,
            "--export",
            request.export,
            "--hardware-profile",
            request.hardware_profile,
            "--k-total",
            str(request.k_total),
            "--top-k",
            str(request.top_k),
        ]
        if request.synergy:
            command.append("--synergy")
            command.extend(["--synergy-combos", str(request.synergy_combos)])
            command.extend(["--synergy-threshold", str(request.synergy_threshold)])
        command.extend(["--top-n", str(request.top_n), "--focus", request.focus])
        return {"executes": False, "command": command}

    return app


app = create_app()
```
