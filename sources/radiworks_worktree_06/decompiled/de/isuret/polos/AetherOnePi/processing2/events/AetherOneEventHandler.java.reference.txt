    private void analyzeCurrentDatabase() {
        this.analyzeDatabase(this.p.getSelectedDatabase());
        this.p.setAnalysisPointer(null);
    }

    private void analyzeDatabase(String databaseName) {
        try {
            this.p.setGvCounter(0);
            this.p.setGeneralVitality(0);
            this.dataService.refreshDatabaseList();
            List<Rate> rates = this.dataService.findAllBySourceName(databaseName);
            AnalysisResult result = this.analyseService.analyseRateList(rates);
            ArrayList<RateObject> rateObjects = new ArrayList<RateObject>();
            for (RateObject rate : result.getRateObjects()) {
                rate.setNameOrRate(this.cleanRateName(rate.getNameOrRate()));
                rate = this.checkForRecurrence(rate);
                rateObjects.add(rate);
            }
            result.setRateObjects(rateObjects);
            this.p.setAnalysisResult(result);
            this.p.setSelectedDatabase(databaseName);
            WebSocketManager.updateAnalysis();
            this.saveCase();
        }
        catch (IOException e) {
            this.log.error("Error analyzing ", e);
        }
    }

    private void saveGeneralVitality() {
        if (this.p.getAnalysisResult() == null) {
            System.err.println("ERROR, null analysis object");
            return;
        }

            Session lastSession = this.p.getCaseObject().getSessionList().get(this.p.getCaseObject().getSessionList().size() - 1);
            lastSession.setAnalysisResult(this.p.getAnalysisResult());

    private RateObject checkForRecurrence(RateObject rateObject) {
        for (int i = 0; i < this.recurringRateList.size(); ++i) {
            RateObject rate = this.recurringRateList.get(i);
            if (!rate.getNameOrRate().equals(rateObject.getNameOrRate())) continue;
            rate.setRecurring(rate.getRecurring() + 1);
            this.recurringRateList.set(i, rate);
            return rate;
        }
        this.recurringRateList.add(rateObject);
        return rateObject;
    }

    private String cleanRateName(String nameOrRate) {
        if (nameOrRate == null) {
            return "---";
        }
        if (nameOrRate.length() > 100) {
            return nameOrRate.substring(0, 100);
        }
        return nameOrRate;
    }

    @Override
    public void keyPressed(char key) {
        if (this.p.getGuiElements().getCurrentTab().equals("ANALYZE")) {
            if (key == '\n') {
                this.checkGeneralVitality();
                return;
            }
            if (this.p.keyCode == 17) {
                this.analyzeCurrentDatabase();
                return;
            }
            this.navigateThroughHistoricalAnalysis();
        }
        if (this.p.keyEvent.getKeyCode() == 17 && this.p.getGuiElements().getCurrentTab().equals("BROADCAST")) {
            String text = this.getTextFromClipboard();
            if (text == null) {
                return;
            }
            ((Textfield)this.p.getGuiElements().getCp5().get("SIGNATURE")).setText(text);
            ((Textfield)this.p.getGuiElements().getCp5().get("SECONDS")).setText("60");
        }
    }

    public void navigateThroughHistoricalAnalysis() {
        if ((this.p.keyCode == 37 || this.p.keyCode == 39) && this.p.getCaseObject() != null && this.p.getCaseObject().getSessionList() != null && this.p.getCaseObject().getSessionList().size() > 0) {
            if (this.p.getAnalysisPointer() == null) {
                this.p.setAnalysisPointer(this.p.getCaseObject().getSessionList().size() - 1);
            }
            if (this.p.keyCode == 37 && this.p.getAnalysisPointer() > 0) {
                this.p.setAnalysisPointer(this.p.getAnalysisPointer() - 1);
            } else if (this.p.keyCode == 39 && this.p.getAnalysisPointer() < this.p.getCaseObject().getSessionList().size() - 1) {
                this.p.setAnalysisPointer(this.p.getAnalysisPointer() + 1);
            }
            AnalysisResult analysisResult = this.p.getCaseObject().getSessionList().get(this.p.getAnalysisPointer()).getAnalysisResult();
            if (analysisResult != null) {
                this.p.setAnalysisResult(analysisResult);
            }
        }
    }

    private String getTextFromClipboard() {
        String text = (String)this.getFromClipboard(DataFlavor.stringFlavor);
        if (text == null) {
            return "";
        }
        return text;
    }

    private Object getFromClipboard(DataFlavor flavor) {
        try {
            Clipboard clipboard = Toolkit.getDefaultToolkit().getSystemClipboard();
            Transferable contents = clipboard.getContents(null);
            if (contents != null && contents.isDataFlavorSupported(flavor)) {
                return contents.getTransferData(flavor);
            }
        }
        catch (UnsupportedFlavorException e) {
            this.log.trace(e);
        }
        catch (IOException e) {
            this.log.trace(e);
        }
        return null;
    }

    private void checkGeneralVitality() {
        if (this.p.getAnalysisResult() == null) {
            return;
        }
        if (this.p.getGvCounter() == null) {
            this.p.setGvCounter(0);
        }
        if (this.p.getAnalysisResult() != null && this.p.getAnalysisPointer() != null) {
            this.addNewSession();
            this.p.setAnalysisPointer(null);
            this.p.setGvCounter(0);
            return;
        }
        if (this.p.getGvCounter() > 21 || this.p.getGvCounter() > this.p.getAnalysisResult().getRateObjects().size()) {
            this.cleanAnalysisForNewGvCheck();
            this.addNewSession();
            return;
        }
        Integer gv = this.p.checkGeneralVitalityValue();
        if (this.p.getGvCounter() == 0) {
            this.p.setGeneralVitality(gv);
        } else {
            this.setRateGeneralVitality(gv);
        }
        this.p.setGvCounter(this.p.getGvCounter() + 1);
        this.saveGeneralVitality();
    }

    private void addNewSession() {
        try {
            if (this.p.getCaseObject().getSessionList().size() == 0) {
                System.out.println("No case or session object in memory");
                return;
            }
            Session lastSession = this.p.getCaseObject().getSessionList().get(this.p.getCaseObject().getSessionList().size() - 1);
            Session newSession = new Session(lastSession);
            this.p.getCaseObject().getSessionList().add(newSession);
            this.p.setAnalysisResult(new AnalysisResult(this.p.getAnalysisResult()));
            newSession.setAnalysisResult(this.p.getAnalysisResult());
            this.p.saveCase();
        }
        catch (Exception e) {
            e.printStackTrace();
        }
    }

    public void cleanAnalysisForNewGvCheck() {
        this.p.setGvCounter(0);
        for (int iRate = 0; iRate < this.p.getAnalysisResult().getRateObjects().size(); ++iRate) {
            RateObject rateObject = this.p.getAnalysisResult().getRateObjects().get(iRate);
            rateObject.setGv(0);
            rateObject.setRecurringGeneralVitality(0);
        }
    }

    public void setRateGeneralVitality(Integer gv) {
        RateObject r;
        int iRate;
        if (this.p.getGvCounter() > this.p.getAnalysisResult().getRateObjects().size()) {
            return;
        }
        RateObject rateObject = this.p.getAnalysisResult().getRateObjects().get(this.p.getGvCounter() - 1);
        rateObject.setGv(gv);
        HashMap<Integer, Integer> gvOccurrences = new HashMap<Integer, Integer>();
        for (iRate = 0; iRate < this.p.getAnalysisResult().getRateObjects().size(); ++iRate) {
            r = this.p.getAnalysisResult().getRateObjects().get(iRate);
            if (r.getGv() == 0) continue;
            if (gvOccurrences.get(r.getGv()) == null) {
                gvOccurrences.put(r.getGv(), 1);
                continue;
            }
            Integer count = (Integer)gvOccurrences.get(r.getGv());
            gvOccurrences.put(r.getGv(), count + 1);
        }
        for (iRate = 0; iRate < this.p.getAnalysisResult().getRateObjects().size(); ++iRate) {
            r = this.p.getAnalysisResult().getRateObjects().get(iRate);
            if (r.getGv() == 0) {
                r.setRecurringGeneralVitality(0);
                continue;
            }
            if (gvOccurrences.get(r.getGv()) != null) {
                r.setRecurringGeneralVitality((Integer)gvOccurrences.get(r.getGv()));
                continue;
            }
            r.setRecurringGeneralVitality(0);
        }
    }
